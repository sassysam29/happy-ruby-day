let currentIndex = 0;
function nextScreen() {
document.getElementById('screen-welcome').classList.add('hidden');
document.getElementById('screen-content').classList.remove('hidden');
renderNote();
}
function renderNote() {
const container =
document.getElementById('note-container');
if (currentIndex < notes.length) {
container.innerHTML = `
<h3>Note ${currentIndex + 1} of
${notes.length}</h3>
<div class="note-box">
<h4 style="color: #d81b60;
margin-top:0;">${notes[currentIndex].title}</h4>
<p>${notes[currentIndex].text}</p>
</div>
`;
if (currentIndex === notes.length - 1) {
document.getElementById('next-btn').innerText =
"View Complete Letter
💌
";
}
}
}
function nextNote() {
currentIndex++;
if (currentIndex < notes.length) {
renderNote();
} else {
showSummary();
}
}
function showSummary() {
document.getElementById('screen-content').classList.add('hidden');
document.getElementById('screen-summary').classList.remove('hidden');
const fullList = document.getElementById('full-list');
fullList.innerHTML = "";
notes.forEach((note, index) => {
fullList.innerHTML += `
<div class="note-box">
<strong>${index + 1}. ${note.title}</strong>
<p style="margin: 5px 0 0 0; font-size:
15px;">${note.text}</p>
</div>
`;
});
}
function toFinalClosingScreen() {
document.getElementById('screen-summary').classList.add('hidden');
document.getElementById('screen-closing').classList.remove('hidden');
}
function restartExperience() {
currentIndex = 0;
document.getElementById('screen-closing').classList.add('hidden');
document.getElementById('screen-welcome').classList.remove('hidden');
document.getElementById('next-btn').innerText = "Next Note
➔";
}
